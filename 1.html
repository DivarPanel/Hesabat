Google Apps Script Mənbə Kodu (`Code.gs`)

Bu kodu Google Sheets > Extensions > Apps Script bölməsinə yapışdırın və "Deploy as Web App" edin.

Aktiv Web App Endpoint: https://script.google.com/macros/s/AKfycbyp_oDoZ3vznDKLrfHd0-sJiBA0Oi53doZFj_wrJhyecPHGfUO1hqoT21QJuUd-jxL8uQ/exec 
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("Transactions") || ss.getSheetByName("Əməliyyatlar") || ss.getSheets()[0];
  var data = sheet.getDataRange().getValues();
  
  var transactions = [];
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    if (!row[0]) continue;
    
    transactions.push({
      id: "TRX-" + (100 + i),
      date: formatDate(row[0]),
      type: row[1] || "Satış",
      customer: row[2] || "Müştəri",
      item: row[3] || "Məhsul",
      qty: Number(row[4]) || 1,
      price: Number(row[5]) || 0,
      total: Number(row[6]) || (Number(row[4]||1) * Number(row[5]||0)),
      paymentType: row[7] || "Nağd"
    });
  }
  
  return ContentService.createTextOutput(JSON.stringify({ transactions: transactions }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Transactions") || ss.getSheetByName("Əməliyyatlar") || ss.getSheets()[0];
    var data = JSON.parse(e.postData.contents);
    
    sheet.appendRow([
      data.date || new Date(),
      data.type || "Satış",
      data.customer || "",
      data.item || "",
      data.qty || 1,
      data.price || 0,
      data.total || 0,
      data.paymentType || "Nağd"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function formatDate(d) {
  if (d instanceof Date) {
    return Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm");
  }
  return d.toString();
}
